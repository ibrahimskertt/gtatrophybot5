import math
from datetime import datetime, timezone

RELEASE = datetime(2026, 11, 19, tzinfo=timezone.utc)
Z = [-1.644854, -0.674490, 0.0, 0.674490, 1.644854]
PRIOR = {'platinum': 0.55, 'gold': 0.75, 'silver': 0.90, 'bronze': 1.00}

def clamp(v, lo, hi):
    return max(lo, min(v, hi))

def logit(rate):
    r = clamp(rate, 0.01, 99.99)
    p = r / 100.0
    return math.log(p / (1 - p))

def inv_logit(l):
    return 100.0 / (1.0 + math.exp(-l))

def hours(ts):
    iso = ts.rstrip('Z')
    if ts.endswith('Z'):
        iso += '+00:00'
    dt = datetime.fromisoformat(iso)
    return (dt - RELEASE).total_seconds() / 3600.0

def wls(u, y, w):
    swuu = sum(wi * ui * ui for wi, ui in zip(w, u))
    if swuu == 0:
        return 0.0, 0.0
    swuy = sum(wi * ui * yi for wi, ui, yi in zip(w, u, y))
    slope = swuy / swuu
    resid = [yi - slope * ui for ui, yi in zip(u, y)]
    sw = sum(w)
    rms = math.sqrt(sum(wi * ri * ri for wi, ri in zip(w, resid)) / sw) if sw else 0.0
    return slope, rms

def forecast(points, tier, x_t, horizon_days):
    xs, rates = zip(*points)
    x_last, r_last = xs[-1], rates[-1]
    if len(points) < 3 or all(r == rates[0] for r in rates):
        b = PRIOR[tier]
        s_fit = 0.30
    else:
        u = [math.log(1 + x) for x in xs]
        y = [logit(r) for r in rates]
        w = [1 + (x / x_last if x_last > 0 else 0) for x in xs]
        slope, rms = wls(u, y, w)
        s_fit = max(rms, 0.30)
        b = 0.65 * slope + 0.35 * PRIOR[tier]

    damp = 1.0 / (1.0 + math.exp((r_last - 50.0) / 12.0))
    med = logit(r_last) + g * (0.35 + 0.65 * damp)
    rate = inv_logit(med)
    if r_last < 20.0:
        bound = min(99.5, max(r_last * 8.0, r_last + 12.0))
        rate = min(rate, bound)
    lo = max(0.01, r_last * 0.25)
    rate = clamp(rate, lo, 99.99)

import json, os, glob
from model import forecast, hours

def main():
    indir = os.environ.get('MAKEFUN_INPUT_DIR', '/input')
    outdir = os.environ.get('MAKEFUN_OUTPUT_DIR', '/output')
    os.makedirs(outdir, exist_ok=True)
    for path in sorted(glob.glob(os.path.join(indir, '*.in'))):
        with open(path) as f:
            case = json.load(f)
        x_t = hours(case['target'])
        hd = case['horizon_minutes'] / 1440.0
        rows = []
        for idx, tr in enumerate(case['trophies']):
            points = [(hours(rd['at']), rd['rates'][idx]) for rd in case['history'] if rd['rates'][idx] is not None]
            if not points:
                points = [(0.0, 0.01)]
            rows.append({'id': tr['id'], 'q': forecast(points, tr['tier'], x_t, hd)})
        out_path = os.path.join(outdir, case['id'] + '.out')
        with open(out_path, 'w') as f:
            f.write(json.dumps({'trophies': rows}, separators=(',', ':')))
if __name__ == '__main__':
    main()
